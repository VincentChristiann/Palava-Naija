    // what a player has bought: the player can read it, only the server writes it
    match /entitlements/{uid} {
      allow read: if request.auth != null && request.auth.uid == uid;
      allow write: if false;
    }
    // the purchase log: server only
    match /purchases/{ref} {
      allow read, write: if false;
    }
    // State Wars: anyone can read the table, only the server (api/statewars.js) writes it
    match /statewars/{week} {
      allow read: if true;
      allow write: if false;
      match /{rest=**} {
        allow read: if true;
        allow write: if false;
      }
    }

    match /players/{uid} {
      allow read, write: if request.auth != null && request.auth.uid == uid;
    }
    match /boards/{board}/scores/{uid} {
      allow read: if board.matches('[a-z0-9_]{2,32}');
      allow create: if request.auth != null && request.auth.uid == uid
        && board.matches('[a-z0-9_]{2,32}')
        && request.resource.data.score is int
        && request.resource.data.score > 0 && request.resource.data.score < 100000000
        && request.resource.data.name is string && request.resource.data.name.size() <= 16;
      allow update: if request.auth != null && request.auth.uid == uid
        && request.resource.data.score is int
        && request.resource.data.score > resource.data.score
        && request.resource.data.score < 100000000
        && request.resource.data.name is string && request.resource.data.name.size() <= 16;
    }
    match /ghosts/{board}/runs/{rid} {
      allow read: if true;
      allow write: if request.auth != null
        && rid.matches(request.auth.uid + '.*')
        && request.resource.data.uid == request.auth.uid
        && request.resource.data.d.size() < 200000;
    }
