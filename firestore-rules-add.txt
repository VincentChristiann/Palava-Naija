    // what a player has bought: the player can read it, only the server writes it
    match /entitlements/{uid} {
      allow read: if request.auth != null && request.auth.uid == uid;
      allow write: if false;
    }
    // the purchase log: server only
    match /purchases/{ref} {
      allow read, write: if false;
    }
